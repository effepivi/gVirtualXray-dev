314 //----------------------------------------------
315 inline void XRayDetector::updateInternalValues()
316 //----------------------------------------------
317 {
318  // The distance between the source and the detector has changed
319  m_distance_source_detector = (m_detector_position - m_xray_source_centre).length();
320 
321  // Compute the normal vector
322  m_look_at_vector = m_detector_position - m_xray_source_centre;
323  m_look_at_vector.normalize();
324  m_right_vector = m_look_at_vector ^ m_up_vector;
325  m_normal = m_up_vector ^ m_right_vector;
326 }

331  RATIONAL_NUMBER& B,

333  RATIONAL_NUMBER& D)
334 //------------------------------------------------------------------
