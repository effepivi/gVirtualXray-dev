62 inline void XRayBeam::initialise(const char* aFileName,
63  const RATIONAL_NUMBER& anUnit,
64  unsigned int aMergeChannelFlag)
65 //-------------------------------------------------------------
66 {
67  // Empty the current beam spectrum
68  m_beam_spectrum.clear();
69 
70  // Open the file
71  std::ifstream beam_spectrum_stream(aFileName);
72 
73  // The file is not open
74  if (!beam_spectrum_stream.is_open())
75  {
76  throw FileDoesNotExistException(__FILE__, __FUNCTION__, __LINE__, aFileName);
77  }
78 
79  // Load the file
80  SpectrumRecord temp;
81  while (!beam_spectrum_stream.eof())
82  {
83  // Read a record
84  beam_spectrum_stream >> temp;
85 
86  // The record is valid
87  if (!beam_spectrum_stream.eof())
88  {
89  // The number of photons is greater than the threshold
90  if (temp.m_photon_number > m_number_of_photons_lower_threshold)
91  {
92  // Apply the unit
93  temp.m_energy *= anUnit;
94 
95  // Store the data
96  m_beam_spectrum.push_back(temp);
97  }
98  }
99  }
100 
101  // Merge channels
102  if (aMergeChannelFlag > 1)
103  {
104  // Merge channels
105  unsigned int counter(0);
106  temp.m_photon_number = 0;
107  temp.m_energy = 0;
108 
109  // The set of new records
110  std::vector<SpectrumRecord> new_beam_spectrum;
111 
112  // Process all the spectrum
113  for (std::vector<SpectrumRecord>::iterator ite(m_beam_spectrum.begin());
114  ite != m_beam_spectrum.end();
115  ++ite)
116  {
117  // Add the current 'old' bin to the 'new' bin
118  temp.m_photon_number += ite->m_photon_number;
119  temp.m_energy += ite->m_energy * ite->m_photon_number;
120  counter++;
121 
122  // Store the new bin
123  if (counter == aMergeChannelFlag)
124  {
125  // Compute the average the energy
126  temp.m_energy /= temp.m_photon_number;
127 
128  // Add the bin
129  new_beam_spectrum.push_back(temp);
130 
131  // Reset counters
132  counter = 0;
133  temp.m_photon_number = 0;
134  temp.m_energy = 0;
135  }
136  }
137 
138  // Add remaining channels if any
139  if (counter)
140  {
141  temp.m_energy /= temp.m_photon_number;
142  new_beam_spectrum.push_back(temp);
143  }
144 
145  // Store the new spectrum
146  m_beam_spectrum = new_beam_spectrum;
147  }
148 
149  // Get the min and max energy
150  double min_energy(FLT_MAX);
151  double max_energy(-FLT_MAX);
152  for (std::vector<SpectrumRecord>::const_iterator ite(m_beam_spectrum.begin());
153  ite != m_beam_spectrum.end();
154  ++ite)
155  {
156  double photon_number(ite->m_photon_number);
157  double photon_energy(ite->m_energy);
158  double bin_energy(photon_energy * photon_number);
159 
160  if (min_energy > bin_energy) min_energy = bin_energy;
161  if (max_energy < bin_energy) max_energy = bin_energy;
162  }
163 
164  // Normalise the beam energy
165  double range(max_energy - min_energy);
166  for (std::vector<SpectrumRecord>::iterator ite(m_beam_spectrum.begin());
167  ite != m_beam_spectrum.end();
168  ++ite)
169  {
170  ite->m_photon_number /= range;
171  }
172 }
