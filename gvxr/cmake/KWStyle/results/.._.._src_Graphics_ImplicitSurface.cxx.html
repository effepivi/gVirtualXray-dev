145 void ImplicitSurface::voxelise(const Vec3ui& aNumberOfVoxels,
146 TAB    const VEC3& aCenter,
147 TAB    const VEC3& aVoxelSize)
148 //-----------------------------------------------------------
149 {
150 #ifdef HAS_ITK
151 TAB,IND VEC3 half_voxel_size(aVoxelSize.getX() / 2.0,
152 TAB aVoxelSize.getY() / 2.0,
153 TAB aVoxelSize.getZ() / 2.0);
154
155 TAB,IND VEC3 half_volume_size(half_voxel_size.getX() * aNumberOfVoxels.getX(),
156 TAB half_voxel_size.getY() * aNumberOfVoxels.getY(),
157 TAB half_voxel_size.getZ() * aNumberOfVoxels.getZ());
158
159 TAB,IND VEC3 offset(aCenter - half_volume_size + half_voxel_size);
160
161 TAB,IND // Get the current size of the volume
162 TAB,IND unsigned int previous_width(0);
163 TAB,IND unsigned int previous_height(0);
164 TAB,IND unsigned int previous_depth(0);
165
166 TAB,IND if (m_p_volume.IsNotNull())
167 TAB,IND {
168 TAB,IND previous_width  = m_p_volume->GetBufferedRegion().GetSize()[0];
169 TAB,IND previous_height = m_p_volume->GetBufferedRegion().GetSize()[1];
170 TAB,IND previous_depth  = m_p_volume->GetBufferedRegion().GetSize()[2];
171 TAB,IND }
172
173 TAB,IND // The size has changed
174 TAB,IND if (previous_width != aNumberOfVoxels.getX() ||
175 TAB previous_height != aNumberOfVoxels.getY() ||
176 TAB previous_depth != aNumberOfVoxels.getZ())
177 TAB {
178 TAB m_p_volume = InputVolumeType::New();
179 TAB m_p_threshold_filter = BinaryThresholdType::New();
180 TAB m_p_iso_surface_extracter = IsoSurfaceExtracterType::New();
181
182 TAB // Initialise the threshold filter
183 TAB m_p_threshold_filter->SetInput(m_p_volume);
184 TAB m_p_threshold_filter->SetOutsideValue(0);
185 TAB m_p_threshold_filter->SetInsideValue(1);
186
187 TAB // Initialise the isosurface extracter
188 TAB const OutputVoxelDataType object_value(static_cast<OutputVoxelDataType>(1));
189 TAB m_p_iso_surface_extracter->SetObjectValue(object_value);
190 TAB m_p_iso_surface_extracter->SetInput(m_p_threshold_filter->GetOutput());
191
192 TAB // Change the size of volume
193 TAB InputVolumeType::IndexType start_index;
194 TAB start_index[0] = 0;
195 TAB start_index[1] = 0;
196 TAB start_index[2] = 0;
197
198 TAB InputVolumeType::SizeType volume_size;
199 TAB volume_size[0] = aNumberOfVoxels.getX();
200 TAB volume_size[1] = aNumberOfVoxels.getY();
201 TAB volume_size[2] = aNumberOfVoxels.getZ();
202
203 TAB itk::ImageRegion<3> region(start_index, volume_size);
204 TAB m_p_volume->SetRegions(region);
205 TAB m_p_volume->Allocate();
206 TAB }
207
208 TAB double spacing[3];
209 TAB spacing[0] = aVoxelSize.getX();
210 TAB spacing[1] = aVoxelSize.getY();
211 TAB spacing[2] = aVoxelSize.getZ();
212 TAB m_p_volume->SetSpacing(spacing);
213
214 TAB double origin[3];
215 TAB origin[0] = offset.getX();
216 TAB origin[1] = offset.getY();
217 TAB origin[2] = offset.getZ();
218 TAB m_p_volume->SetOrigin(origin);
219
220 TAB VEC3 voxel_center;
221
222 TAB InputVolumeType::IndexType voxel_index;
223
224 TAB RATIONAL_NUMBER r(0);
225
226 TAB VEC3 point_1;
227 TAB VEC3 point_2;
228
229 TAB VEC3 v1;
230 TAB VEC3 v2;
231
232 LEN,TAB itk::ImageRegionIterator<InputVolumeType> voxel_ite(m_p_volume, m_p_volume->GetLargestPossibleRegion());
233 TAB voxel_ite.GoToBegin();
234 TAB for (unsigned int z(0); z < aNumberOfVoxels.getZ(); ++z)
235 TAB {
236 TAB voxel_center.setZ(offset.getZ() + z * aVoxelSize.getZ());
237
238 TAB for (unsigned int y(0); y < aNumberOfVoxels.getY(); ++y)
239 TAB {
240 TAB voxel_center.setY(offset.getY() + y * aVoxelSize.getY());
241
242 TAB for (unsigned int x(0); x < aNumberOfVoxels.getX(); ++x, ++voxel_ite)
243 TAB {
244 TAB RATIONAL_NUMBER voxel_value(0);
245
246 TAB voxel_center.setX(offset.getX() + x * aVoxelSize.getX());
247
248 TAB // Process all the points
249 LEN,TAB for (std::vector<std::pair<DensityFunctionParameterType, VEC3> >::const_iterator point_ite(m_control_point_set.begin());
250 TAB point_ite != m_control_point_set.end();
251 TAB ++point_ite)
252 TAB {
253 TAB r = (voxel_center - point_ite->second).length();
254 TAB voxel_value += evaluate(r, point_ite->first);
255 TAB }
256
257 TAB // Process all the lines
258 LEN,TAB for (std::vector<std::pair<DensityFunctionParameterType, std::pair<VEC3, VEC3> > >::const_iterator line_ite(m_control_line_set.begin());
259 TAB line_ite != m_control_line_set.end();
260 TAB ++line_ite)
261 TAB {
262 TAB point_1 = line_ite->second.first;
263 TAB,IND point_2 = line_ite->second.second;
264
265 LEN,TAB const RATIONAL_NUMBER length_squared(pow((point_1 - point_2).length(), 2.0));
266 TAB,IND if (fabs(length_squared) < EPSILON)
267 TAB {
268 TAB r = (voxel_center - point_2).length();
269 TAB }
270 TAB else
271 TAB {
272 LEN,TAB const RATIONAL_NUMBER normalised_dot_product(((voxel_center - point_1).dotProduct(point_2 - point_1)) / length_squared);
273 TAB if (normalised_dot_product < 0.0)
274 TAB {
275 TAB r = (voxel_center - point_1).length();
276 TAB }
277 TAB else if (normalised_dot_product > 1.0)
278 TAB {
279 TAB r = (voxel_center - point_2).length();
280 TAB }
281 TAB else
282 TAB {
283 LEN,TAB const VEC3 projection(point_1 + normalised_dot_product * (point_2 - point_1));
284 TAB r = (voxel_center - projection).length();
285 TAB }
286 TAB }
287
288 TAB voxel_value += evaluate(r, line_ite->first);
289 TAB,IND }
290
291 TAB,IND itk::Point<double, 3> voxel_position;
292 TAB voxel_position[0] = voxel_center.getX();
293 TAB voxel_position[1] = voxel_center.getY();
294 TAB voxel_position[2] = voxel_center.getZ();
295 TAB m_p_volume->TransformPhysicalPointToIndex(voxel_position, voxel_index);
296 TAB m_p_volume->SetPixel(voxel_index, voxel_value);
297 TAB }
298 TAB }
299 TAB }
300 #else // HAS_ITK
301 LEN,TAB throw Exception(__FILE__, __FUNCTION__, __LINE__, "Not implemented, you have to enable ITK using CMake.");
302 #endif // HAS_ITK
303 }
