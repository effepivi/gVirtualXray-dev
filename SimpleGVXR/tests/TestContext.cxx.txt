extern std::map<int, std::unique_ptr<SimpleGVXR::Window> > g_p_window_set;      // The set of windows
extern std::vector<int> g_p_window_id_stack;           // The stack of window IDs


TEST(EGLContext, PassContextBetweenThreads)
{
    std::mutex mutex;
    std::condition_variable cv;

    bool thread_a_done = false;

    std::atomic<bool> a_success(false);
    std::atomic<bool> b_success(false);

    std::thread thread_a([&]
    {
        // Create an OpenGL context
#ifdef HAS_OPENGL
#ifdef HAS_EGL
        createWindow(-1, 0,
            "EGL",
            3, 2,
            0); // Half-precision floating-point numbers is enough
#else // HAS_EGL
#ifdef __APPLE__
        // OpenGL 4.2 and above are not supported on MacOS (at least on my compute with MacOS Big Sur)
        createWindow(-1, 0,
            "OPENGL",
            4, 1,
            0); // Half-precision floating-point numbers is enough
#else // __APPLE__
        createWindow(-1, 0,
            "OPENGL",
            4, 6,
            0); // Half-precision floating-point numbers is enough
#endif // __APPLE__
#endif // HAS_EGL
#else // HAS_OPENGL
        createWindow(-1, 0, "OPENMP");
#endif // HAS_OPENGL

        useAutoUnbindContext(true);

        GLint major = 0;
        glGetIntegerv(GL_MAJOR_VERSION, &major);

        a_success = major > 0;

        g_p_window_set[g_p_window_id_stack.back()]->getContext().unbindContext();


        {
            std::lock_guard<std::mutex> lock(mutex);
            thread_a_done = true;
        }

        cv.notify_one();
    });

    std::thread thread_b([&]
    {
        {
            std::unique_lock<std::mutex> lock(mutex);

            cv.wait(
                lock,
                [&]
                {
                    return thread_a_done;
                });
        }

        g_p_window_set[g_p_window_id_stack.back()]->getContext().bindContext();
        GLint major = 0;
        glGetIntegerv(GL_MAJOR_VERSION, &major);

        b_success = major > 0;
    });

    thread_a.join();
    thread_b.join();

    EXPECT_TRUE(a_success);
    EXPECT_TRUE(b_success);
}
