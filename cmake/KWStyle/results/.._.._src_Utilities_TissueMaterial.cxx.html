167 //-------------------------------------------------------
168 double TissueMaterial::getDensity(short aHounsfieldValue)
169 //-------------------------------------------------------
170 {
171     double density(0);
172
173     if (aHounsfieldValue <= -1000)
174     {
175         density = 1.21e-3;
176     }
177     else if (-1000 <= aHounsfieldValue && aHounsfieldValue < -98)
178     {
179 LEN         // interpolate mass densities of air (1.21e-3 g/cm3) and adipose tissue 3 (0.93 g/cm3)
180 LEN         density = 1.21e-3 + (aHounsfieldValue - -1000) * (0.93 - 1.21e-3) / (-98 - -1000);
181     }
182     else if (-98 <= aHounsfieldValue && aHounsfieldValue <= 14)
183     {
184         // Use of Eq. 21 in W Schneider et al 2000
185         density = 1.018 + 0.893e-3 * aHounsfieldValue;
186     }
187     else if (14 < aHounsfieldValue && aHounsfieldValue < 23)
188     {
189         density = 1.03;
190     }
191     else if (23 <= aHounsfieldValue && aHounsfieldValue <= 100)
192     {
193         // Use of Eq. 23 in W Schneider et al 2000
194         density = 1.003 + 1.169e-3 * aHounsfieldValue;
195     }
196     else if (100 <= aHounsfieldValue)
197     {
198         // Use of Eq. 19 in W Schneider et al 2000
199         density = 1.017 + 0.592e-3 * aHounsfieldValue;
200     }
201
202     return (density * (g / cm3));
203 }
