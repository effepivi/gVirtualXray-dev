143 inline bool PolygonMesh::useVBO(int aTypeOfVBO, int aTypeOfPrimitive)
144 //-------------------------------------------------------------------
145 {
146     // Create the VBO if needed
147     if (!m_vbo.get())
148         {
149         m_vbo.reset(createVBO());
150         }
151
152     // There is no data to store in a VBO
153     if (m_number_of_vertices && (getVertexNormalNumber() ||
154             getFaceNormalNumber()))
155         {
156         // Create a VBO if possible
157         updateVBO(aTypeOfVBO, aTypeOfPrimitive);
158         }
159
160     return (m_vbo->isReady());
161 }

302 //-------------------------------------------
303 inline void PolygonMesh::computeBoundingBox()
304 //-------------------------------------------
305 {
306     // There are vertices
307     if (m_number_of_vertices)
308     {
309         m_min_corner = getVertex(0);
310         m_max_corner = getVertex(0);
311
312         for (unsigned int i(1); i < m_number_of_vertices; ++i)
313         {
314             VEC3 vertex(getVertex(i));
315
316             if (m_min_corner.getX() > vertex.getX())
317                 m_min_corner.setX(vertex.getX());
318
319             if (m_max_corner.getX() < vertex.getX())
320                 m_max_corner.setX(vertex.getX());
321
322             if (m_min_corner.getY() > vertex.getY())
323                 m_min_corner.setY(vertex.getY());
324
325             if (m_max_corner.getY() < vertex.getY())
326                 m_max_corner.setY(vertex.getY());
327
328             if (m_min_corner.getZ() > vertex.getZ())
329                 m_min_corner.setZ(vertex.getZ());
330
331             if (m_max_corner.getZ() < vertex.getZ())
332                 m_max_corner.setZ(vertex.getZ());
333         }
334
335         // Compute diagonal size
336         m_bounding_box_diagonal = (m_min_corner - m_max_corner).length();
337     }
338 }

341 //----------------------------------------------------------------
342 inline void PolygonMesh::applyScale(const RATIONAL_NUMBER& aScale)
343 //----------------------------------------------------------------
344 {
345     // There are vertices
346     if (m_number_of_vertices)
347     {
348         for (unsigned int i(0); i < m_number_of_vertices; ++i)
349         {
350             VEC3 vertex(getVertex(i) * aScale);
351             setVertex(i, vertex);
352         }
353
354         // Update the bounding box
355         m_min_corner *= aScale;
356         m_max_corner *= aScale;
357         m_bounding_box_diagonal *= aScale;
358
359         // Create the VBO if needed
360         if (!m_vbo.get())
361         {
362             m_vbo.reset(createVBO());
363         }
364
365         // Update the VBO
366         if (m_vbo->isReady())
367         {
368             updateVBO(m_vbo->getBufferUsageHint(), m_vbo->getPrimitiveType());
369         }
370     }
371 }

374 //----------------------------------------------------------------
375 inline void PolygonMesh::translate(const VEC3& aTranslationVector)
376 //----------------------------------------------------------------
377 {
378     if (m_number_of_vertices)
379         {
380         for (unsigned int i(0); i < m_number_of_vertices; i++)
381             {
382             VEC3 vertex(getVertex(i) + aTranslationVector);
383             setVertex(i, vertex);
384             }
385
386         m_min_corner += aTranslationVector;
387         m_max_corner += aTranslationVector;
388
389         // Create the VBO if needed
390         if (!m_vbo.get())
391             {
392             m_vbo.reset(createVBO());
393             }
394
395         // Update the VBO
396         if (m_vbo->isReady())
397             {
398             updateVBO(m_vbo->getBufferUsageHint(), m_vbo->getPrimitiveType());
399             }
400         }

405 inline void PolygonMesh::moveToCentre()
406 //-------------------------------------
407 {
408     VEC3 translation_vector(m_min_corner + (m_max_corner - m_min_corner) / 2.0);
409     translation_vector *= -1.0;

415 //-------------------------------------------
416 inline void PolygonMesh::inverNormalVectors()
417 //-------------------------------------------
418 {
419     for (std::vector<RATIONAL_NUMBER>::iterator ite(
420                             m_p_vertex_normal_set.begin());
421             ite != m_p_vertex_normal_set.end();
422             ++ite)
423         {
424         *ite = -(*ite);
425         }
426
427
428     for (std::vector<RATIONAL_NUMBER>::iterator ite(
429                         m_p_face_normal_set.begin());
430             ite != m_p_face_normal_set.end();
431             ++ite)
432         {
433         *ite = -(*ite);
434         }
435 }
