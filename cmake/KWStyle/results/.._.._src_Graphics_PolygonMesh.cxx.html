431 //------------------------------------------------------------------
432 void PolygonMesh::loadSTLDataFromStream(const char* apInputData,
433 TAB bool aMoveToCentreFlag,
434 TAB bool anAutoComputeNormalFlag,
435 TAB bool aPrintDebugInfoFlag,
436 TAB bool aCreateVBOFlag,
437 TAB RATIONAL_NUMBER aScale,
438 TAB int aBufferUsageHing)
439 //------------------------------------------------------------------
440 {
441 TAB // Get the number of triangles in the file header
442 LEN,TAB // (the header is made of 80 bytes followded by the number of triangles (4 bytes))
443 TAB char const* p_input_data(apInputData);
444 TAB p_input_data += 80;
445
446 TAB int triangle_number;
447
448 TAB memcpy(&triangle_number, p_input_data, sizeof(int));
449 TAB p_input_data += sizeof(int);
450
451 TAB // Set the kind of polygon
452 TAB m_polygon_type = GL_TRIANGLES;
453
454 TAB // Set the size of arrays to store the vertices and normal vectors
455 TAB m_external_data_flag = false;
456 TAB m_number_of_vertices = triangle_number * 3;
457 TAB GLfloat* p_vertex_set(new GLfloat[m_number_of_vertices * 3]);
458 TAB m_vertex_data_type = GL_FLOAT;
459 TAB m_p_vertex_set = p_vertex_set;
460
461 TAB m_p_face_normal_set.resize(m_number_of_vertices * 3);
462 TAB m_p_vertex_normal_set.resize(m_number_of_vertices * 3);
463
464 TAB
465 TAB // Read the data
466 TAB for (unsigned int i(0); i < getFaceNumber(); i++)
467 TAB {
468 TAB // Read normals
469 TAB memcpy(&m_p_face_normal_set[9 * i], p_input_data, sizeof(float) * 3);
470 TAB p_input_data += sizeof(float) * 3;
471
472 TAB,ESP // Copy one normal per vertex*
473 TAB if (!anAutoComputeNormalFlag)
474 TAB {
475 TAB for (unsigned int k=1; k<3; k++)
476 TAB {
477 TAB m_p_face_normal_set[9 * i + 3 * k]     = m_p_face_normal_set[9 * i];
478 TAB m_p_face_normal_set[9 * i + 3 * k + 1] = m_p_face_normal_set[9 * i + 1];
479 TAB m_p_face_normal_set[9 * i + 3 * k + 2] = m_p_face_normal_set[9 * i + 2];
480
481 TAB m_p_vertex_normal_set[9 * i + 3 * k]     = m_p_face_normal_set[9 * i];
482 TAB m_p_vertex_normal_set[9 * i + 3 * k + 1] = m_p_face_normal_set[9 * i + 1];
483 TAB m_p_vertex_normal_set[9 * i + 3 * k + 2] = m_p_face_normal_set[9 * i + 2];
484 TAB }
485 TAB }
486
487 TAB // Read vertices
488 TAB memcpy(&p_vertex_set[9 * i], p_input_data, sizeof(float) * 3 * 3);
489 TAB p_input_data += sizeof(float) * 3 * 3;
490
491 TAB // Skip data
492 TAB p_input_data += 2;
493 TAB }
494
495 TAB // Compute the normal vectors using the cross product
496 TAB if (anAutoComputeNormalFlag)
497 TAB {
498 TAB computeNormalVectors();
499 TAB normaliseNormals();
500 TAB }
501
502 TAB // Change scale and compute bounding box
503 TAB applyScale(aScale);
504 TAB computeBoundingBox();
505
506 TAB // Move the object to the center
507 TAB if (aMoveToCentreFlag)
508 TAB {
509 TAB moveToCentre();
510 TAB }
511
512 TAB // Display extra debug info
513 TAB if (aPrintDebugInfoFlag)
514 TAB {
515 TAB std::cout << "file_name:\t" << m_filename <<
516 TAB "\tnb_faces:\t" << getFaceNumber() <<
517 TAB "\tnb_vertices:\t" << m_number_of_vertices <<
518 LEN,TAB "\tbounding_box (in cm):\t(" << m_min_corner.getX() / cm << ", " << m_min_corner.getY() / cm << ", " << m_min_corner.getZ() / cm << ")\t(" <<
519 LEN,TAB m_max_corner.getX() / cm << ", " << m_max_corner.getY() / cm << ", " << m_max_corner.getZ() / cm << ")" << std::endl;
520 TAB }
521 TAB
522 TAB // Create the VBO
523 TAB if (aCreateVBOFlag)
524 TAB {
525 TAB updateVBO(aBufferUsageHing, m_polygon_type);
526 TAB }
527 }

530 //------------------------------
531 void PolygonMesh::writeSTLFile()
532 //------------------------------
533 {
534 TAB if (m_polygon_type != GL_TRIANGLES)
535 TAB {
536 LEN,TAB throw Exception(__FILE__, __FUNCTION__, __LINE__, "Cannot write a STL file, mesh is not made of triangles.");
537 TAB }
538
539 TAB if (!getTriangleNumber())
540 TAB {
541 LEN,TAB throw Exception(__FILE__, __FUNCTION__, __LINE__, "Cannot write a STL file, there is no triangle to save.");
542 TAB }
543
544 TAB if (!m_filename.size())
545 TAB {
546 LEN,TAB throw Exception(__FILE__, __FUNCTION__, __LINE__, "Cannot write a STL file, not file name provided.");
547 TAB }
548
549 TAB // Open the file in writing more
550 LEN     std::ofstream output_stream(m_filename.c_str(), std::ios::out | std::ios::binary);
551
552 TAB // The file is not open
553 TAB if (!output_stream.is_open())
554 TAB {
555 LEN,TAB throw Exception(__FILE__, __FUNCTION__, __LINE__, "Cannot open the STL file in writing mode");
556 TAB }
557
558 TAB // The header is made of 80 bytes
559 TAB char temp(0);
560 TAB for (unsigned int i(0); i < 80; ++i)
561 TAB {
562 TAB output_stream.write(&temp, 1);
563 TAB }
564
565 TAB // The number of triangles
566 TAB int triangle_number(getTriangleNumber());
567 TAB output_stream.write((char*) &triangle_number, sizeof(triangle_number));
568
569 TAB // Read the data
570 TAB float* p_vertex_set((float*) m_p_vertex_set);
571
572 TAB // Use an index
573 TAB if (getIndexNumber())
574 TAB {
575 TAB for (unsigned int i(0); i < getFaceNumber(); i++)
576 TAB {
577 TAB int id0(getIndex(i * 3 + 0));
578 TAB int id1(getIndex(i * 3 + 1));
579 TAB int id2(getIndex(i * 3 + 2));
580
581 TAB VEC3 v0(getVertex(id0));
582 TAB VEC3 v1(getVertex(id1));
583 TAB VEC3 v2(getVertex(id2));
584
585 TAB VEC3 n0(getVertexNormal(id0));
586 TAB VEC3 n1(getVertexNormal(id1));
587 TAB VEC3 n2(getVertexNormal(id2));
588 TAB VEC3 normal(n0 + n1 + n2);
589 TAB normal.normalise();
590
591 TAB float temp[3];
592 TAB temp[0] = normal.getX();
593 TAB temp[1] = normal.getY();
594 TAB temp[2] = normal.getZ();
595 TAB output_stream.write((char*) temp, sizeof(float) * 3);
596
597 TAB temp[0] = v0.getX();
598 TAB temp[1] = v0.getY();
599 TAB temp[2] = v0.getZ();
600 TAB output_stream.write((char*) temp, sizeof(float) * 3);
601
602 TAB temp[0] = v1.getX();
603 TAB temp[1] = v1.getY();
604 TAB temp[2] = v1.getZ();
605 TAB output_stream.write((char*) temp, sizeof(float) * 3);
606
607 TAB temp[0] = v2.getX();
608 TAB temp[1] = v2.getY();
609 TAB temp[2] = v2.getZ();
610 TAB output_stream.write((char*) temp, sizeof(float) * 3);
611
612 TAB // Write data
613 TAB char dummy(0);
614 TAB output_stream.write(&dummy, 1);
615 TAB output_stream.write(&dummy, 1);
616 TAB }
617 TAB }
618 TAB // Do not use an index
619 TAB else
620 TAB {
621 TAB for (unsigned int i(0); i < getFaceNumber(); i++)
