445 //-------------------------------------------------------------------------
446 void XRayRenderer::computeSinogram(MATRIX4& aModellingTransformationMatrix,
447                                    VEC3 aRotationCenter,
448                                    unsigned int aNumberOfAngle,
449                                    RATIONAL_NUMBER anAngleOffset)
450 //-------------------------------------------------------------------------
451 {
452     if (anAngleOffset == 0)
453     {
454 LEN         throw Exception(__FILE__, __FUNCTION__, __LINE__, "The number of projections is zero.");
455     }
456
457 LEN     VEC3 pixel_size(m_p_detector->getSizeInUnitOfLength().getX() / m_p_detector->getNumberOfPixels().getX(),
458             anAngleOffset,
459 LEN             m_p_detector->getSizeInUnitOfLength().getY() / m_p_detector->getNumberOfPixels().getY());
460
461     if (std::abs(pixel_size.getZ()) < EPSILON)
462     {
463         pixel_size[2] = 1.0;
464     }
465
466     // Convert the size in mm * degree * mm
467     pixel_size[0] /= mm;
468     pixel_size[1] /= 1.0;
469     pixel_size[2] /= mm;
470
471     #ifdef HAS_ITK
472     // Compute the size of a pixel
473     ImageType::SpacingType itk_pixel_size;
474     itk_pixel_size[0] = pixel_size.getX();
475     itk_pixel_size[1] = pixel_size.getY();
476     itk_pixel_size[2] = pixel_size.getZ();
477
478
479     // Create a new image
480     ImageType::IndexType start;
481     start[0] = 0;
482     start[1] = 0;
483     start[2] = 0;
484
485     ImageType::SizeType sinogram_size;
486     sinogram_size[0] = m_p_detector->getNumberOfPixels().getX();
487     sinogram_size[1] = aNumberOfAngle;
488     sinogram_size[2] = m_p_detector->getNumberOfPixels().getY();
489
490     ImageType::RegionType sinogram_region;
491     sinogram_region.SetSize(sinogram_size);
492     sinogram_region.SetIndex(start);
493
494     // Allocate memory
495     m_p_sinogram_image->SetRegions(sinogram_region);
496
497     m_p_sinogram_image->Allocate();
498
499     // Set the pixel size
500     m_p_sinogram_image->SetSpacing(itk_pixel_size);
501
502 #else // HAS_ITK
503
504     // Allocate the memory
505     m_sinogram_image = Image<PixelType>(
506             m_p_detector->getNumberOfPixels().getX(),
507             aNumberOfAngle,
508             m_p_detector->getNumberOfPixels().getY());
509
510     // Set the pixel size
511     m_sinogram_image.setSpacing(pixel_size);
512
513 #endif // HAS_ITK
514
515     // Compute the translation vector
516     VEC3 translation_vector(-aRotationCenter);
517
518     // Compute the sinogram
519     for (unsigned int angle(0); angle < aNumberOfAngle; ++angle)
520     {
521         MATRIX4 transformation_matrix;
522
523         // Translate the sample
524         transformation_matrix.translate(-translation_vector);
525
526         // Rotate the sample
527 LEN         transformation_matrix.rotate(anAngleOffset * angle, m_p_detector->getUpVector());
528
529 IND ********// Translate the sample
530         transformation_matrix.translate(translation_vector);
531
532 IND ********// Apply the modelling transformation
533 LEN         transformation_matrix = transformation_matrix * aModellingTransformationMatrix;
534
535         // Compute the X-ray image
536 IND ********computeImage(transformation_matrix);
537
538 IND ********// Load the FBO in the main memory
539         getFBO(XRAY_DETECTOR_ENERGY_FLUENCE_FBO_ID);
540
541 IND ********// Get the destination pointer
542 #ifdef HAS_ITK
543 IND ********PixelType* p_pixel_set(m_p_sinogram_image->GetBufferPointer());
544 #else // HAS_ITK
545 IND ********PixelType* p_pixel_set(m_sinogram_image.getRawData());
546 #endif // HAS_ITK
547
548 IND ********// Convert the data into a sinogram
549         PixelType* p_input_data(m_temp_raw_data.getRawData());
550 IND ********RATIONAL_NUMBER total_input_energy(m_p_xray_beam->getTotalEnergy());
551 IND ********RATIONAL_NUMBER scaling_factor(
552 LEN                 RATIONAL_NUMBER(m_p_detector->getNumberOfPixels().getX()) / (m_p_detector->getSizeInUnitOfLength().getX() / cm)
