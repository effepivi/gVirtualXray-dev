125 //-------------------------------------------------------------
126 inline void XRayBeam::initialise(const char* aFileName,
127 TAB  const RATIONAL_NUMBER& anUnit,
128 TAB  unsigned int aMergeChannelFlag)
129 //-------------------------------------------------------------
130 {
131 TAB,IND // Empty the current beam spectrum
132 TAB,IND m_beam_spectrum.clear();
133 TAB,IND
134 TAB,IND // Open the file
135 TAB,IND std::ifstream beam_spectrum_stream(aFileName);
136 TAB
137 TAB // The file is not open
138 TAB if (!beam_spectrum_stream.is_open())
139 TAB {
140 TAB throw FileDoesNotExistException(__FILE__, __FUNCTION__, __LINE__, aFileName);
141 TAB,IND }
142
143 TAB,IND // Load the file
144 TAB,IND SpectrumRecord temp;
145 TAB,IND while (!beam_spectrum_stream.eof())
146 TAB,IND {
147 TAB,IND // Read a record
148 TAB,IND beam_spectrum_stream >> temp;
149 TAB,IND
150 TAB,IND // The record is valid
151 TAB,IND if (!beam_spectrum_stream.eof())
152 TAB,IND {
153 TAB,IND // The number of photons is greater than the threshold
154 TAB if (temp.m_photon_number > m_number_of_photons_lower_threshold)
155 TAB,IND {
156 TAB,IND // Apply the unit
157 TAB,IND temp.m_energy *= anUnit;
158 TAB,IND
159 TAB,IND // Store the data
160 TAB,IND m_beam_spectrum.push_back(temp);
161 TAB,IND }
162 TAB,IND }
163 TAB,IND }
164 TAB,IND
165 TAB,IND // Merge channels
166 TAB,IND if (aMergeChannelFlag > 1)
167 TAB,IND {
168 TAB,IND // Merge channels
169 TAB,IND unsigned int counter(0);
170 TAB,IND temp.m_photon_number = 0;
171 TAB,IND temp.m_energy = 0;
172 TAB,IND
173 TAB,IND // The set of new records
174 TAB,IND std::vector<SpectrumRecord> new_beam_spectrum;
175 TAB
176 TAB // Process all the spectrum
177 TAB for (std::vector<SpectrumRecord>::iterator ite(m_beam_spectrum.begin());
178 TAB ite != m_beam_spectrum.end();
179 TAB ++ite)
180 TAB {
181 TAB,IND // Add the current 'old' bin to the 'new' bin
182 TAB temp.m_photon_number += ite->m_photon_number;
183 TAB,IND temp.m_energy += ite->m_energy * ite->m_photon_number;
184 TAB,IND counter++;
185 TAB,IND
186 TAB,IND // Store the new bin
187 TAB,IND if (counter == aMergeChannelFlag)
188 TAB {
189 TAB // Compute the average the energy
190 TAB temp.m_energy /= temp.m_photon_number;
191 TAB,IND
192 TAB,IND // Add the bin
193 TAB,IND new_beam_spectrum.push_back(temp);
194 TAB,IND
195 TAB,IND // Reset counters
196 TAB,IND counter = 0;
197 TAB,IND temp.m_photon_number = 0;
198 TAB,IND temp.m_energy = 0;
199 TAB,IND }
200 TAB,IND }
201 TAB,IND
202 TAB,IND // Add remaining channels if any
203 TAB,IND if (counter)
204 TAB,IND {
205 TAB,IND temp.m_energy /= temp.m_photon_number;
206 TAB new_beam_spectrum.push_back(temp);
207 TAB,IND }
208
209 TAB,IND // Store the new spectrum
210 TAB,IND m_beam_spectrum = new_beam_spectrum;
211 TAB,IND }
212 TAB,IND
213 TAB,IND // Get the min and max energy
214 TAB,IND double min_energy(FLT_MAX);
215 TAB,IND double max_energy(-FLT_MAX);
216 TAB,IND for (std::vector<SpectrumRecord>::const_iterator ite(m_beam_spectrum.begin());
217 TAB ite != m_beam_spectrum.end();
218 TAB ++ite)
219 TAB {
220 TAB,IND double photon_number(ite->m_photon_number);
221 TAB,IND double photon_energy(ite->m_energy);
222 TAB,IND double bin_energy(photon_energy * photon_number);
223 TAB,IND
224 TAB,IND if (min_energy > bin_energy) min_energy = bin_energy;
225 TAB,IND if (max_energy < bin_energy) max_energy = bin_energy;
226 TAB }
227 TAB
228 TAB // Normalise the beam energy
229 TAB double range(max_energy - min_energy);
230 TAB for (std::vector<SpectrumRecord>::iterator ite(m_beam_spectrum.begin());
231 TAB ite != m_beam_spectrum.end();
232 TAB ++ite)
233 TAB {
234 TAB ite->m_photon_number /= range;
235 TAB,IND }
236 }
