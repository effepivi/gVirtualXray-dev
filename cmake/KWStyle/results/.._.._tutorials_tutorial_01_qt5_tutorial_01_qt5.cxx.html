356 void GLWidget::paintGL()
357 //----------------------
358 IND {
359     // Clear the buffers
360     glClear(GL_COLOR_BUFFER_BIT | GL_DEPTH_BUFFER_BIT);

362     // Add the current shader to the shader stack
363     pushShaderProgram();

366     g_display_shader.enable();
367     GLint shader_id(g_display_shader.getProgramHandle());
368
369     // Check the status of OpenGL and of the current FBO
370     checkFBOErrorStatus(__FILE__, __FUNCTION__, __LINE__);
371     checkOpenGLErrorStatus(__FILE__, __FUNCTION__, __LINE__);

373     // A handle for shader resources

384     checkOpenGLErrorStatus(__FILE__, __FUNCTION__, __LINE__);
385
386     // Store the current transformation matrices
387     pushModelViewMatrix();
388     pushProjectionMatrix();
389
390     // Translate the 1st object
391     g_current_modelview_matrix *= MATRIX4::buildTranslationMatrix(
