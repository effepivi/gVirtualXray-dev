362 //----------------------
363 void GLWidget::paintGL()
364 //----------------------
365 {
366     // Clear the buffers
367     glClear(GL_COLOR_BUFFER_BIT | GL_DEPTH_BUFFER_BIT);
368
369     // Add the current shader to the shader stack
370     pushShaderProgram();
371
372     // Enable the shader
373     g_display_shader.enable();
374     GLint shader_id(g_display_shader.getProgramHandle());
375
376     // Check the status of OpenGL and of the current FBO
377     checkFBOErrorStatus(__FILE__, __FUNCTION__, __LINE__);
378     checkOpenGLErrorStatus(__FILE__, __FUNCTION__, __LINE__);
379
380     // A handle for shader resources
381     GLuint handle(0);
382
383     // Upload the projection matrix
384     handle = glGetUniformLocation(shader_id, "g_projection_matrix");
385     glUniformMatrix4fv(handle, 1, GL_FALSE, g_current_projection_matrix.get());
386     checkOpenGLErrorStatus(__FILE__, __FUNCTION__, __LINE__);
387
388     // Upload the modelview matrix
389     handle = glGetUniformLocation(shader_id, "g_modelview_matrix");
390     glUniformMatrix4fv(handle, 1, GL_FALSE, g_current_modelview_matrix.get());
391     checkOpenGLErrorStatus(__FILE__, __FUNCTION__, __LINE__);
392
393     // Store the current transformation matrices
394     pushModelViewMatrix();
395     pushProjectionMatrix();
396
397     // Rotate the 3D scene
398     g_current_modelview_matrix *= g_scene_rotation_matrix;
399
400     // Wireframe mode
401     handle = glGetUniformLocation(shader_id, "g_wireframe");
402     glUniform1i(handle, false);
403     checkOpenGLErrorStatus(__FILE__, __FUNCTION__, __LINE__);
404
405     // The first object is the inner cube
406     handle = glGetUniformLocation(shader_id, "g_inner_surface");
407     glUniform1i(handle, 0);
408     checkOpenGLErrorStatus(__FILE__, __FUNCTION__, __LINE__);
409
410     // Display the source
411     g_xray_detector.displaySource();
412
413     // Display the detector with the corresponding X-ray image
414     g_xray_simulator.display();
415
416     // Rotate the 1st object
417     g_current_modelview_matrix *= g_sample_rotation_matrix;
418
419     // Apply the change to the shader program
420     applyModelViewMatrix();
421
422     // Wireframe mode
423     handle = glGetUniformLocation(shader_id, "g_wireframe");
424     glUniform1i(handle, g_wireframe);
425     checkOpenGLErrorStatus(__FILE__, __FUNCTION__, __LINE__);
426
427     // The first object is the inner cube
428     handle = glGetUniformLocation(shader_id, "g_inner_surface");
429     glUniform1i(handle, 1);
430     checkOpenGLErrorStatus(__FILE__, __FUNCTION__, __LINE__);
431
432     // Display the 1st object
433     if (!g_wireframe)
434     {
435         g_polygon_mesh_1.display();
436     }
437     else
438     {
439         g_polygon_mesh_1.displayWireFrame();
440     }
441
442     // Store the state of alpha blending
443     if (!g_wireframe)
444     {
445         pushEnableDisableState(GL_BLEND);
446         glEnable(GL_BLEND);
447         glDepthMask (GL_FALSE);
448         glBlendFunc (GL_SRC_ALPHA, GL_ONE_MINUS_SRC_ALPHA);
449     }
450
451     // The second object is the outer cube
452     glUniform1i(handle, 0);
453     checkOpenGLErrorStatus(__FILE__, __FUNCTION__, __LINE__);
454
455     // Display the 2nd object
456     if (!g_wireframe)
457     {
458         g_polygon_mesh_2.display();
459     }
460     else
461     {
462         g_polygon_mesh_2.displayWireFrame();
463     }
464
465     // Restore the state of alpha blending
466     if (!g_wireframe)
467     {
468         popEnableDisableState();
469         glDepthMask (GL_TRUE);
470     }
471
472     // Restore the current transformation matrix
473     popModelViewMatrix();
474     popProjectionMatrix();
475
476     // Disable the shader and restore the previous shader from the stack
477     popShaderProgram();
478
479     // Check the status of OpenGL and of the current FBO
480     checkFBOErrorStatus(__FILE__, __FUNCTION__, __LINE__);
481     checkOpenGLErrorStatus(__FILE__, __FUNCTION__, __LINE__);
482
483     animate();
484 }

488 MBF void GLWidget::animate()
489 //---------------------
490 {

504 }

507 //--------------------------------------------
508 void GLWidget::resizeGL(int width, int height)
509 //--------------------------------------------
510 {
511     // Avoid a division by zero
512     if (height == 0)
513     {
514         // Prevent divide by 0
515         height = 1;
516     }
517
518     int x(0), y(0), w(width), h(height);
519
520     // Store the width and height of the window
521     g_window_width = width;
522     g_window_height = height;
523
524     // Compute the aspect ratio of the size of the window
525     double screen_aspect_ratio(double(g_window_width) / double(g_window_height));
526
527     // Update the viewport
528     glViewport(x, y, w, h);
529
530     // Set up the projection matrix (g_current_projection_matrix)
531 LEN     loadPerspectiveProjectionMatrix(45.0, screen_aspect_ratio, 0.1 * cm, 5000.0 * cm);
532
533     // Set up the modelling-viewing matrix (g_current_modelview_matrix)
534     loadLookAtModelViewMatrix(0.0 * cm, 0.0 * cm, g_zoom,
535             0.0, 0.0, 0.0,
536             0.0, 1.0, 0.0);
537 }

614 //---------------
615 {
616     // Initialise the shaders
617     char* p_vertex_shader(0);
618     char* p_fragment_shader(0);
619
