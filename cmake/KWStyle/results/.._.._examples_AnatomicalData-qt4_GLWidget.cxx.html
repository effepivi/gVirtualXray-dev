308 //----------------------
309 void GLWidget::paintGL()
310 //----------------------
311 {
312     // Clear the buffers
313     glClear(GL_COLOR_BUFFER_BIT | GL_DEPTH_BUFFER_BIT);
314
315     // Display the 3D elements
316     glViewport(0, 0, width(), height());
317
318     // Store the current transformation matrix
319     pushModelViewMatrix();
320     pushProjectionMatrix();
321
322     double screen_aspect_ratio(double(width()) / double(height()));
323
324 LEN     loadPerspectiveProjectionMatrix(g_initial_fovy, screen_aspect_ratio, g_initial_near, g_initial_far);
325
326     loadLookAtModelViewMatrix(0.0, 0.0, m_zoom,
327             0.0, 0.0, 0.0,
328             0.0, 1.0, 0.0);
329
330     // Rotation
331     if (m_current_angle != m_next_angle_to_process)
332     {
333         switch (m_acquisition_type)
334         {
335 IND,IND ********case ROTATE_SAMPLE:
336 IND ************{
337                 Matrix4x4<GLfloat> P4_rotation_matrix;
338 LEN                 P4_rotation_matrix *= Matrix4x4<GLfloat>::buildTranslationMatrix( m_P4_translation_vector);
339 LEN                 P4_rotation_matrix *= Matrix4x4<GLfloat>::buildRotationMatrix(-m_angle_step, g_initial_detector_up_vector);
340 LEN                 P4_rotation_matrix *= Matrix4x4<GLfloat>::buildTranslationMatrix(-m_P4_translation_vector);
341
342                 Matrix4x4<GLfloat> P5_rotation_matrix;
343 LEN                 P5_rotation_matrix *= Matrix4x4<GLfloat>::buildTranslationMatrix( m_P5_translation_vector);
344 LEN                 P5_rotation_matrix *= Matrix4x4<GLfloat>::buildRotationMatrix(-m_angle_step, g_initial_detector_up_vector);
345 LEN                 P5_rotation_matrix *= Matrix4x4<GLfloat>::buildTranslationMatrix(-m_P5_translation_vector);
346
347                 m_P4_rotation_matrix *= P4_rotation_matrix;
348                 m_P5_rotation_matrix *= P5_rotation_matrix;
349 IND ************}
350             break;
351
352 IND,IND ********case ROTATE_SOURCE_DETECTOR:
353 IND ************{
354                 Matrix4x4<GLfloat> rotation_matrix;
355 LEN                 rotation_matrix *= Matrix4x4<GLfloat>::buildRotationMatrix(m_angle_step, g_initial_detector_up_vector);
356
357 LEN                 VEC3 source_position(  rotation_matrix * m_xray_detector.getXraySourceCentre());
358 LEN                 VEC3 detector_position(rotation_matrix * m_xray_detector.getDetectorPosition());
359
360                 // Set the source position
361                 m_xray_detector.setXrayPointSource(source_position);
362
363                 // Use a parallel source
364                 m_xray_detector.setParallelBeam();
365
366                 // Set the detector position
367                 m_xray_detector.setDetectorPosition(detector_position);
368 IND ************}
369             break;
370 IND ********}
371     }
372
373     // Update the current angle
374     m_current_angle = m_next_angle_to_process;
375     m_next_angle_to_process += m_angle_step;
376
377     // Stop the acquisition
378     if (m_next_angle_to_process > 180.0)
379 IND ****{
380         if (m_swap_patient)
381 IND ********{
382             if (m_current_patient == P4)
383             {
384                 usePatient(P5);
385 IND ************}
386             else
387 IND ************{
388                 usePatient(P4);
389 IND ************}
390 IND ********}
391         else
392 IND ********{
393             reset();
394 IND ********}
395
396         m_swap_patient = !m_swap_patient;
397 IND ****}
398
399     // Update the X-ray image
400     updateXRayImage();
401     checkOpenGLErrorStatus(__FILE__, __FUNCTION__, __LINE__);
402
403     // Draw the 3D scene
404     render();
405     checkOpenGLErrorStatus(__FILE__, __FUNCTION__, __LINE__);
406
407     // Draw the help message
408 #ifdef HAS_FREETYPE
409     if (m_display_help)
410 IND ****{
411         displayHelp();
412
413         // Check the status of OpenGL and of the current FBO
414         checkFBOErrorStatus(__FILE__, __FUNCTION__, __LINE__);
415         checkOpenGLErrorStatus(__FILE__, __FUNCTION__, __LINE__);
416 IND ****}
417 #endif
418
419     // Restore the current transformation matrix
420     popModelViewMatrix();
421     popProjectionMatrix();
422
423     // Make sure all the OpenGL code is done
424     glFinish();
425
426     animate();
427 }

580 SPL     delete [] p_vertex_shader;     p_vertex_shader = 0;
581 SPL     delete [] p_fragment_shader; p_fragment_shader = 0;
582

590 }
