219 //-------------------------------
220 void OpenGLWindow::initializeGL()
221 //-------------------------------
222 {
223     try
224     {
225         GLenum glew_error_1(glewInit());
226         if (glew_error_1 != GLEW_OK)
227         {
228             std::stringstream error_message;
229             error_message << "Cannot initialize GLEW: " <<
230                 glewGetErrorString(glew_error_1);
231             throw Exception(__FILE__, __FUNCTION__, __LINE__,
232                 error_message.str());
233         }
234
235         glClearColor(.1f, .1f, .1f, 1);
236         glEnable(GL_DEPTH_TEST);
237
238         // Choose smooth shading
239         glShadeModel(GL_SMOOTH);
240
241         // Use colour and texture coordinate interpolation
242         glHint(GL_PERSPECTIVE_CORRECTION_HINT, GL_NICEST);
243
244         // Use antialiased polygons
245         glHint(GL_POLYGON_SMOOTH_HINT, GL_NICEST);
246
247         // Use antialiasing
248         glEnable(GL_POLYGON_SMOOTH);
249
250         //glEnable(GL_MULTISAMPLE);
251         glHint(GL_MULTISAMPLE_FILTER_HINT_NV, GL_NICEST);
252
253         // Add the geometry to the X-ray renderer
254         std::string prefix(m_executable_path);
255         if (prefix.length())
256             {
257             prefix += "/";
258             }
259     
260         prefix += "..";
261
262         std::string stl_filename(prefix);
263         stl_filename += "/data/welsh-dragon-small.stl";
264
265         loadSTLFile(stl_filename);
266
267         loadDetector();
268         loadSource();
269         loadXRaySimulator();
270         loadShader();
271
272         m_xray_renderer.addInnerSurface(&m_polygon_data);
273
274         // Rotate the sample
275         m_sample_rotation_matrix.rotate( 180, VEC3(0, 1, 0));
276         m_sample_rotation_matrix.rotate( 90, VEC3(0, 0, 1));
277         m_sample_rotation_matrix.rotate( 90, VEC3(0, 1, 0));
278
279         // Rotate the scene
280         m_scene_rotation_matrix.rotate(-110, VEC3(0, 1, 0));
281
282         // Enable the stereo if possible
283         // Set the stereo parameters
284         m_stereo_helper.setIntraocularDistance(g_initial_intraocular_distance);
285         g_p_intraocular_distance_slider->value(g_initial_intraocular_distance);
286         g_p_intraocular_distance_slider->range(
287             0,
288             g_initial_intraocular_distance + g_initial_intraocular_distance /
289                 1.5);
290         m_stereo_helper.setFieldOfViewY(g_initial_fovy);
291         m_stereo_helper.setNear(g_initial_near);
292         m_stereo_helper.setFar(g_initial_far);
293         m_stereo_helper.setScreenProjectionPlane(
294             g_initial_screen_projection_plane);
295         m_stereo_helper.enable();
296
297         // The stereo cannot be activated
298         if (!m_stereo_helper.isActive())
299         {
300             // Disable widgets
301             g_p_stereo_button->deactivate();
302             g_p_intraocular_distance_slider->deactivate();
303         }
304         else
305         {
306             g_p_stereo_button->value(true);
307         }
308     }
309     catch (const std::exception& e)
310     {
311         const char* p_error_message(e.what());
312         fl_alert(p_error_message);
313         //exit (EXIT_FAILURE);
314     }
315 }

328 //-----------------------
329 void OpenGLWindow::draw()
330 //-----------------------
331 {
332     try
333         {
334         if (m_first_time)
335             {
336             initializeGL();
337             reshape(w(), h());
338             m_first_time = false;
339             }
340
341         // Compute the X-ray image
342         updateXRayImage();
343
344         // Get the current draw buffer if needed
345         GLint back_buffer(0);
346         glGetIntegerv(GL_DRAW_BUFFER, &back_buffer);
347
348         if (m_stereo_helper.isActive())
349             {
350             m_stereo_helper.swapEye();
351
352             // Clear the buffers
353             glClear(GL_COLOR_BUFFER_BIT | GL_DEPTH_BUFFER_BIT);
354
355             // Display the 3D scene
356             render();
357
358             // Make sure all the OpenGL code is done
359             glFinish();
360
361             m_stereo_helper.swapEye();
362             }
363
364         // Clear the buffers
365         glClear(GL_COLOR_BUFFER_BIT | GL_DEPTH_BUFFER_BIT);
366
367         // Display the 3D scene
368         render();
369
370         // Make sure all the OpenGL code is done
371         glFinish();
372
373         // Restore the draw buffer if needed
374         glDrawBuffer(back_buffer);
375         }
376     catch (const std::exception& e)
377         {
378         std::cerr << e.what() << std::endl;
379         }
380 }

383 //-----------------------------------
384 int OpenGLWindow::handle(int anEvent)
385 //-----------------------------------
386 {
387     switch (anEvent)
388         {
389         case FL_FOCUS:
390         case FL_UNFOCUS:
391             return (1);
392
393         case FL_KEYBOARD:
394             {
395             int key = Fl::event_key();
396             switch (key)
397                 {
398                 case FL_Left:
399                     setObjectRotationAngle(-m_rotation_increment);
400                     redraw();
401                     return (1);
402
403                 case FL_Right:
404                     setObjectRotationAngle(+m_rotation_increment);
405                     redraw();
406                     return (1);
407
408                 case ' ':
409                     m_animating = !m_animating;
410                     m_is_xray_image_up_to_date = false;
411                     return (1);
412                 }
413             }
414             break;
415             
416
417         case FL_PUSH:
418
419             m_use_arc_ball = true;
420
421             m_last_x_position = m_current_x_position = Fl::event_x();
422
423             if ( Fl::event_button() == FL_LEFT_MOUSE)
424                 {
425                 m_last_y_position = m_current_y_position = Fl::event_y();
426                 }
427             else
428                 {
429                 m_last_y_position = m_current_y_position = h() - Fl::event_y();
430                 }
431             
432             return (1);
433
434         case FL_RELEASE:
435             m_use_arc_ball = false;
436             return (1);
437
438         case FL_DRAG:
439             if (Fl::event_button() == FL_LEFT_MOUSE)
440                 {
441                 m_current_x_position = Fl::event_x();
442                 m_current_y_position = Fl::event_y();
443                 // left click rotates environment
444                 computeRotation(m_scene_rotation_matrix);
445                 m_is_xray_image_up_to_date = false;
446                 redraw();
447                 return (1);
448                 }
449
450             if (Fl::event_button() == FL_RIGHT_MOUSE)
451                 {
452                 m_current_x_position = Fl::event_x();
453                 m_current_y_position = h() - Fl::event_y();
454                 // right click rotates object
455                 computeRotation(m_sample_rotation_matrix);
456                 m_is_xray_image_up_to_date = false;
457                 redraw();
458                 return (1);
459                 }
460
461             if (Fl::event_button() == FL_MIDDLE_MOUSE)
462                 {
463                 std::cout << "MIDDLE BUTTON." << std::endl;
464                 m_current_x_position = Fl::event_x();
465                 m_current_y_position = h() - Fl::event_y();
466                 // middle click rotates x-ray detector
467                 computeRotation(m_detector_rotation_matrix);
468                 m_xray_detector.setRotationMatrix(m_detector_rotation_matrix);
469                 m_is_xray_image_up_to_date = false;
470                 redraw();
471                 return (1);
472                 }
473             break;
474
475         case FL_MOUSEWHEEL:
476             m_zoom += float(Fl::event_dy()) * cm;
477             if (m_zoom > g_zoom_slider->maximum())
478                 {
479                 m_zoom = g_zoom_slider->maximum();
480                 }
481             
482             if (m_zoom < g_zoom_slider->minimum())
483                 {
484                 m_zoom = g_zoom_slider->minimum();
485                 }
486             
487             g_zoom_slider->value(m_zoom);
488             reshape(w(), h());
489             return (1);
490         }
491
492     return (Fl_Gl_Window::handle(anEvent));
493 }

496 //-------------------------
497 void OpenGLWindow::render()
498 //-------------------------
499 {
500     try
501         {
502         // Enable the shader
503         pushShaderProgram();
504         m_display_shader.enable();
505         GLint shader_id(m_display_shader.getProgramHandle());
506
507         // Check the status of OpenGL and of the current FBO
508         checkFBOErrorStatus(__FILE__, __FUNCTION__, __LINE__);
509         checkOpenGLErrorStatus(__FILE__, __FUNCTION__, __LINE__);
510
511         // Store the current attributes
512         pushEnableDisableState(GL_LIGHT0);
513         pushEnableDisableState(GL_LIGHTING);
514
515         // Store the current transformation matrix
516         pushModelViewMatrix();
517
518         // Rotate the sample
519         g_current_modelview_matrix *= m_scene_rotation_matrix;
520
521         GLuint handle(0);
522
523         VEC3 light_position(0, 0, m_zoom);
524         handle = glGetUniformLocation(shader_id, "light_position");
525         glUniform4f(
526             handle,
527             light_position.getX(),
528             light_position.getY(),
529             light_position.getZ(),
530             1.0);
531         checkOpenGLErrorStatus(__FILE__, __FUNCTION__, __LINE__);
532
533         handle = glGetUniformLocation(shader_id, "g_projection_matrix");
534         glUniformMatrix4fv(
535             handle,
536             1,
537             GL_FALSE,
538             g_current_projection_matrix.get());
539         checkOpenGLErrorStatus(__FILE__, __FUNCTION__, __LINE__);
540
541         handle = glGetUniformLocation(shader_id, "g_modelview_matrix");
542         glUniformMatrix4fv(
543             handle,
544             1,
545             GL_FALSE,
546             g_current_modelview_matrix.get());
547         checkOpenGLErrorStatus(__FILE__, __FUNCTION__, __LINE__);
548
549         MATRIX4 normal_matrix(g_current_modelview_matrix);
550         handle = glGetUniformLocation(shader_id, "g_normal_matrix");
551         glUniformMatrix3fv(handle, 1, GL_FALSE, normal_matrix.get3x3());
552         checkOpenGLErrorStatus(__FILE__, __FUNCTION__, __LINE__);
553
554         GLint lighting(1);
555         if (!m_display_wireframe)
556             {
557             lighting = 1;
558             }
559         else
560             {
561             lighting = 0;
562             }
563         handle = glGetUniformLocation(shader_id,"g_use_lighting");
564         glUniform1iv(handle, 1, &lighting);
565         checkOpenGLErrorStatus(__FILE__, __FUNCTION__, __LINE__);
566
567         // Display the sample
568
569         // Set the colour of the sample
570         {
571         const GLfloat material_ambient[]  = {0.19225,  0.0, 0.0, 1.0};
572         const GLfloat material_diffuse[]  = {0.50754,  0.0, 0.0, 1.0};
573         const GLfloat material_specular[] = {0.508273, 0.0, 0.0, 1.0};
574         const GLfloat material_shininess = 50.2;
575
576         handle = glGetUniformLocation(shader_id, "material_ambient");
577         glUniform4fv(handle, 1, &material_ambient[0]);
578         checkOpenGLErrorStatus(__FILE__, __FUNCTION__, __LINE__);
579
580         handle = glGetUniformLocation(shader_id, "material_diffuse");
581         glUniform4fv(handle, 1, &material_diffuse[0]);
582         checkOpenGLErrorStatus(__FILE__, __FUNCTION__, __LINE__);
583
584         handle = glGetUniformLocation(shader_id, "material_specular");
