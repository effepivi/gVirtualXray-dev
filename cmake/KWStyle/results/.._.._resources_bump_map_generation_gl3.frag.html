237 //calculate the height of the bump
238 //------------------------
239 float computeBumpValue(vec3 object_position)
240 //------------------------
241 {
242 TAB float noise_value;
243 TAB
244 TAB if(g_has_thin_bumps == 1)
245 TAB {
246 TAB noise_value = noise(object_position*2)*0.5+0.5;
247 TAB noise_value *= 0.5;
248 TAB }
249 TAB else
250 TAB {
251 TAB noise_value = noise(object_position*0.5)*0.5+0.5;
252 TAB }
253 TAB
254 TAB //lower the noise
255 TAB float bump_value = noise_value*0.3;
256 TAB
257 TAB if(g_has_tendon == 1)
258 TAB {
259 TAB //create perturbation using noise and fractal sum
260 TAB float perturbation = noise_value + fractalSum(object_position);
261 TAB
262 TAB //lower the perturbation
263 TAB perturbation *= PERTURBATION_COEF;
264 TAB
265 TAB //angle of the point to the center
266 LEN,TAB float theta = atan(object_position.y,object_position.x+length(object_position.xy));
267  
268 TAB float horizontal = clamp(abs(dot(normalize(g_normal),vec3(0,0,1))), 0., 1.);
269 TAB
270 TAB //reduce bump from noise
271 TAB bump_value *= horizontal*horizontal*horizontal;
272 TAB
273 TAB //Creates lines disturbed by fractal sum and noise
274 TAB,OPS bump_value+=0.1*(1-horizontal)*
275 TAB (1-(pow(abs(sin(5*LINE_FREQ*(theta+perturbation))),14)+
276 TAB 0.6*pow(abs(sin(7*LINE_FREQ*(theta+perturbation))),10))+
277 TAB 0.4*pow(abs(sin(9*LINE_FREQ*(theta+perturbation))),8));
278 TAB }
279 TAB
280 TAB if(g_is_lung == 1)
281 TAB {
282 TAB //create perturbation using fractal sum
283 TAB float perturbation = fractalSum(object_position);
284 TAB
285 TAB //lower the perturbation
286 TAB perturbation *= PERTURBATION_COEF;
287 TAB
288 TAB //change the global position
289 TAB perturbation += g_gradient_tab[0].z*0.05;
290 TAB
291 LEN,TAB vec3 rotated_for_right_line_position = (g_right_line_matrix*vec4(object_position,1.)).xyz;
292 LEN,TAB vec3 rotated_for_side_lines_position = (g_side_lines_matrix*vec4(object_position,1.)).xyz;
293 TAB
294 TAB float height;
295 TAB
296 TAB //if the fragment is on the right, the lung has a central line
297 TAB if((object_position.x - g_min_corner.x)/(g_max_corner.x - g_min_corner.x)>0.5)
298 TAB {
299 TAB,IND //The height is 0 where the line is
300 LEN,TAB height = clamp(abs(LINE_SHARP*(rotated_for_right_line_position.z - g_min_corner.z)/(g_max_corner.z - g_min_corner.z)-
301 TAB LINE_SHARP*(LINE_POS+perturbation)),0.,1.);
302 TAB,IND
303 TAB,IND //add another change to the position of the line
304 TAB perturbation += g_gradient_tab[0].y*0.05+0.25;
305 TAB,IND
306 TAB,IND //draw the line on the side
307 LEN,TAB height *= clamp(abs(LINE_SHARP*(rotated_for_side_lines_position.z - g_min_corner.z)/(g_max_corner.z - g_min_corner.z)-
308 TAB LINE_SHARP*(LINE_POS+perturbation)),0.,1.);
309 TAB,IND
310 TAB,IND }
311 TAB,IND //if not, it has only a line on the side
312 TAB,IND else
313 TAB,IND {
314 TAB,IND //add another change to the position of the line and draw it higher
315 TAB perturbation -= g_gradient_tab[0].z*0.05-0.2;
316 TAB,IND
317 TAB,IND //draw the line on the side
318 LEN,TAB height = abs(LINE_SHARP*(rotated_for_side_lines_position.z - g_min_corner.z)/(g_max_corner.z - g_min_corner.z)-
319 TAB LINE_SHARP*(LINE_POS+perturbation));
320 TAB,IND }
321 TAB,IND
322 TAB,IND //reduce bump from noise
323 TAB,IND bump_value *= 0.6;
324 TAB,IND
325 TAB,IND //add the line
326 TAB,IND bump_value += clamp(height,0.,1.);
327 TAB,IND
328 TAB,IND //square to make it sharper
329 TAB,IND bump_value = 1-(1-bump_value)*(1-bump_value);
330 TAB,IND }
331 TAB,IND return bump_value;
