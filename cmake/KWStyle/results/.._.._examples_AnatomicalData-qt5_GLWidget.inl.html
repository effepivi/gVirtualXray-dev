100 //---------------------------------------------------
101 inline void GLWidget::resizeGL(int width, int height)
102 //---------------------------------------------------
103 {
104     // Avoid a division by zero
105     if (height == 0)
106         {
107         // Prevent divide by 0
108         height = 1;
109         }
110
111     int x(0), y(0), w(width), h(height);
112
113     // Update the viewport
114     glViewport(x, y, w, h);
115
116     m_text_2D_projection_matrix = buildOrthoProjectionMatrix(0,
117         width,
118         height,
119         0,
120         -1.0,
121         1.0);
122 }

147 //-------------------------------------------------------
148 inline void GLWidget::wheelEvent(QWheelEvent *wheelEvent)
149 //-------------------------------------------------------
150 {
151     // Scrolling along the Y-axis
152     if (std::abs(wheelEvent->delta()) > EPSILON)
153         {
154         // Stop using the arc ball
155         m_use_arc_ball = false;
156
157         // Change the zoom
158         m_zoom += 0.1 * wheelEvent->delta() * cm;
159
160         // Update the projection matrix
161         resizeGL(width(), height());
162         }
163 }

235 //-----------------------------------------------------------
236 inline Vec3<GLfloat> GLWidget::getArcballVector(int x, int y)
237 //-----------------------------------------------------------
238 {
239     Vec3<GLfloat> P(2.0 * float(x) / float(width()) - 1.0,
240         2.0 * float(y) / float(height()) - 1.0,
241         0);
242
243     P.setY(-P.getY());
244
245     float OP_squared = P.getX() * P.getX() + P.getY() * P.getY();
246     if (OP_squared <= 1.0)
247         {
248         P.setZ(sqrt(1.0 - OP_squared));  // Pythagore
249         }
250     else
251         {
252         P.normalise();  // nearest point
253         }
254
255     return (P);
256 }
