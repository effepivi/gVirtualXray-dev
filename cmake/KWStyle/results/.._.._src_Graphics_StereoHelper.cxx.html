115 //-------------------------
116 void StereoHelper::update()
117 //-------------------------
118 {
119 TAB,IND const double degree_to_radian(PI / 180.0);
120
121 TAB,IND // Sets top of frustum based on fovy and near clipping plane
122 TAB double top = m_near * tan(degree_to_radian * m_fovy / 2.0);
123 LEN,TAB,IND double right = m_screen_aspect_ratio * top;                             //sets right of frustum based on aspect ratio
124 LEN,TAB double frustum_shift = (m_intraocular_distance / 2.0) * m_near / m_screen_projection_plane;
125
126 TAB,IND m_left_eye_frustum[0] = -right + frustum_shift;
127 TAB,IND m_left_eye_frustum[1] =  right + frustum_shift;
128 TAB,IND m_left_eye_frustum[2] = -top;
129 TAB,IND m_left_eye_frustum[3] =  top;
130
131 TAB,IND double left_camera_translation(m_intraocular_distance / 2.0);
132
133 LEN,TAB m_left_eye_projection_matrix = buildFrustumProjectionMatrix(m_left_eye_frustum[0],
134 TAB m_left_eye_frustum[1],
135 TAB m_left_eye_frustum[2],
136 TAB m_left_eye_frustum[3],
137 TAB m_near,
138 TAB m_far);
139
140 LEN,TAB m_left_eye_projection_matrix *= MATRIX4::buildTranslationMatrix(left_camera_translation, 0, 0);
141
142 TAB m_right_eye_frustum[0] = -right - frustum_shift;
143 TAB m_right_eye_frustum[1] =  right - frustum_shift;
144 TAB m_right_eye_frustum[2] = -top;
145 TAB m_right_eye_frustum[3] =  top;
146
147 TAB double right_camera_translation(-left_camera_translation);
148
149 LEN,TAB m_right_eye_projection_matrix = buildFrustumProjectionMatrix(m_right_eye_frustum[0],
150 TAB m_right_eye_frustum[1],
151 TAB m_right_eye_frustum[2],
152 TAB m_right_eye_frustum[3],
153 TAB m_near,
154 TAB m_far);
155
156 LEN,TAB m_right_eye_projection_matrix *= MATRIX4::buildTranslationMatrix(right_camera_translation, 0, 0);
157
158 TAB m_matrices_are_up_to_date = true;
159 }

162 //--------------------------
163 void StereoHelper::swapEye()
164 //--------------------------
165 {
166 TAB,IND // Change the eye
167 TAB,IND switch (m_current_eye)
168 TAB,IND {
169 TAB,IND case LEFT_EYE:
170 TAB m_current_eye = RIGHT_EYE;
171 TAB break;
172
173 TAB case RIGHT_EYE:
174 TAB m_current_eye = LEFT_EYE;
175 TAB break;
176
177 TAB default:
178 TAB if (m_stereo_currently_in_use)
179 TAB {
180 TAB,IND m_current_eye = RIGHT_EYE;
181 TAB,IND }
182 TAB,IND else
183 TAB,IND {
184 TAB,IND m_current_eye = MONO;
185 TAB,IND }
186 TAB,IND break;
187 TAB,IND }
188
189 TAB,IND // Change the back buffer and the projection matrix
190 TAB,IND switch (m_current_eye)
191 TAB,IND {
192 TAB,IND case LEFT_EYE:
193 TAB g_current_projection_matrix = getLeftEyeProjectionMatrix();
194 TAB if (m_stereo_currently_in_use)
195 TAB {
196 TAB,IND glDrawBuffer(GL_BACK_LEFT);
197 TAB,IND }
198 TAB,IND break;
199
200 TAB,IND case RIGHT_EYE:
201 TAB g_current_projection_matrix = getRightEyeProjectionMatrix();
202 TAB if (m_stereo_currently_in_use)
203 TAB {
204 TAB,IND glDrawBuffer(GL_BACK_RIGHT);
205 TAB,IND }
206 TAB,IND break;
207
208 TAB,IND case MONO:
209 TAB default:
210 TAB glDrawBuffer(GL_BACK);
211 TAB break;
212 TAB }
213 }

216 //-------------------------
217 bool StereoHelper::enable()
218 //-------------------------
219 {
220 TAB,IND // Stereo is disable
221 TAB,IND if (!m_stereo_currently_in_use)
222 TAB,IND {
223 TAB,IND // Check the stereo
224 TAB,IND GLboolean can_use_stereo;
225 TAB,IND glGetBooleanv(GL_STEREO, &can_use_stereo);
226
227 TAB,IND // It is enable on the system
228 TAB,IND if (can_use_stereo)
229 TAB,IND {
230 TAB,IND // Enable the use of stereo
231 TAB,IND m_stereo_currently_in_use = true;
232 TAB,IND }
233 TAB,IND // It is not enable on the system
234 TAB,IND else
235 TAB,IND {
236 TAB,IND disable();
237 TAB,IND }
238 TAB,IND }
239
240 TAB,IND return (m_stereo_currently_in_use);
241 }
