227 LEN //--------------------------------------------------------------------------------
228 template<typename T> Matrix4x4<T> Matrix4x4<T>::buildRotationMatrix(T anAngle,
229 TAB T x, T y, T z)
230 LEN //--------------------------------------------------------------------------------
231 {
232 TAB,IND Vec3<T> rotation_axis(Vec3<T>(x, y, z).normal());
233 TAB,IND double angle(gVirtualXRay::PI * anAngle / 180.0);
234 TAB double c(cos(angle));
235 TAB double s(sin(angle));
236 TAB double one_minus_cos(1.0 - c);
237 TAB double xy(rotation_axis.getX() * rotation_axis.getY());
238 TAB double xz(rotation_axis.getX() * rotation_axis.getZ());
239 TAB double yz(rotation_axis.getY() * rotation_axis.getZ());
240
241 TAB Matrix4x4<T> rotation_matrix(
242 TAB rotation_axis.getX() * rotation_axis.getX() * (1.0 - c) + c,
243 TAB xy * one_minus_cos + rotation_axis.getZ() * s,
244 TAB xz * one_minus_cos - rotation_axis.getY() * s,
245 TAB 0.0,
246
247 TAB xy * one_minus_cos - rotation_axis.getZ() * s,
248 TAB rotation_axis.getY() * rotation_axis.getY() * one_minus_cos + c,
249 TAB yz * one_minus_cos + rotation_axis.getX() * s,
250 TAB 0.0,
251
252 TAB xz * one_minus_cos + rotation_axis.getY() * s,
253 TAB yz * one_minus_cos - rotation_axis.getX() * s,
254 TAB rotation_axis.getZ() * rotation_axis.getZ() * one_minus_cos + c,
255 TAB 0.0,
256
257 TAB 0.0,
258 TAB 0.0,
259 TAB 0.0,
260 TAB 1.0);
261
262 TAB return (rotation_matrix);
263 }
