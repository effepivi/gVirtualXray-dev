365 //------------------------------------------------------
366 int Shader::loadFile(const std::string& aShaderFilename,
367 TAB  GLchar**& aShaderSource)
368 //------------------------------------------------------
369 {
370 TAB int line_number(0);
371
372 TAB // Open the file
373 TAB std::ifstream input_stream(aShaderFilename.data());
374
375 TAB // Check if the file is open
376 TAB if (!input_stream.is_open())
377 TAB {
378 LEN,TAB throw FileDoesNotExistException(__FILE__, __FUNCTION__, __LINE__, aShaderFilename);
379 TAB }
380 TAB
381 TAB // Count the number of new line characters ('\n')
382 LEN,TAB line_number = count(std::istreambuf_iterator<char>(input_stream), std::istreambuf_iterator<char>(), '\n');
383
384 TAB // Count the last line
385 TAB ++line_number;
386
387 TAB try
388 TAB {
389 TAB,IND // Allocate the memory
390 TAB,IND aShaderSource = new char * [line_number];
391
392 TAB,IND // Initialise the memory
393 TAB std::fill_n(aShaderSource, line_number, static_cast<char*>(0));
394
395
396 TAB // Read the data from the beginning of the file
397 TAB input_stream.seekg(std::ios_base::beg);
398
399 TAB std::string line;
400 TAB for (int allocated_size(0);
401 TAB allocated_size < line_number;
402 TAB ++allocated_size)
403 TAB {
404 TAB,IND // Read the current line
405 TAB,IND getline(input_stream, line);
406
407 TAB,IND // Add the end of line character
408 TAB,IND line += "\n";
409
410 TAB,IND // Allocate the memory to store the line
411 TAB aShaderSource[allocated_size] = new char [line.size() + 1];
412
413 TAB,IND // Copy the line
414 TAB,IND strcpy(aShaderSource[allocated_size], line.c_str());
415 TAB,IND }
416 TAB,IND }
417 TAB,IND // There was a memory allocation error
418 TAB,IND catch (const std::bad_alloc &)
419 TAB {
420 TAB,IND // Release the memory that has been successfully allocated
421 TAB releaseSourceMemory(line_number, aShaderSource);
422
423 TAB,IND // Close the file
424 TAB,IND input_stream.close();
425
426 TAB,IND // Throw an exception
427 TAB throw OutOfMemoryException(__FILE__, __FUNCTION__, __LINE__);
428 TAB,IND }
429
430 TAB,IND // Close the file
431 TAB,IND input_stream.close();
432
433 TAB,IND return (line_number);
434 }

449 TAB,IND shader_handle = glCreateShader(aShaderType);

451 TAB,IND if (!shader_handle)

453 TAB,IND switch (aShaderType)
454 TAB,IND {
455 TAB,IND case GL_VERTEX_SHADER:

459 TAB case GL_FRAGMENT_SHADER:

461 TAB break;

465 TAB break;

532 TAB throw Exception(__FILE__, __FUNCTION__, __LINE__, error_message.str());
