458 //----------------------------------------------
459 inline void XRayDetector::updateInternalValues()
460 //----------------------------------------------
461 {
462 TAB,IND // The distance between the source and the detector has changed
463 LEN,TAB m_distance_source_detector = (m_detector_position - m_xray_source_centre).length();
464
465 TAB,IND // Compute the normal vector
466 TAB m_look_at_vector = m_detector_position - m_xray_source_centre;
467 TAB,IND m_look_at_vector.normalize();
468 TAB,IND m_right_vector = m_look_at_vector ^ m_up_vector;
469 TAB,IND m_normal = m_up_vector ^ m_right_vector;
470 TAB,IND
471 TAB,IND // Update the view along the Y-axis
472 TAB,IND updateFOVY();
473 }

477 inline const void XRayDetector::getPlaneEquation(RATIONAL_NUMBER& A,
478 TAB  RATIONAL_NUMBER& B,
479 TAB  RATIONAL_NUMBER& C,
480 TAB  RATIONAL_NUMBER& D)
481 //------------------------------------------------------------------
482 {
483 TAB,IND // Transform the normal
484 TAB,IND VEC3 normal(m_rotation_matrix * m_normal);
485 TAB,IND
486 TAB,IND // Compute the plane equation
487 TAB,IND A = normal.getX();
488 TAB,IND B = normal.getY();
489 TAB,IND C = normal.getZ();
490 LEN,TAB,IND D = -(A * m_detector_position.getX() + B * m_detector_position.getY() + C * m_detector_position.getZ());
491 }
